📋 Setup Instructions

  1. Open Google Sheets and create a new spreadsheet named Shield Lunch Orders.
  2. Create three sheets inside it: Menus, Students, and Orders.
  3. Go to Extensions → Apps Script, replace all code with the script below, then click Deploy → New deployment → Web App → set Execute as: Me and Who has access: Anyone.
  4. Copy the Web App URL and paste it into the APPS_SCRIPT_URL constant at the top of this file.
  5. Open Admin → 🍽 Meals, set your menus and click Save Changes — this writes them to the Menus sheet.
  6. Fill in the Students sheet with Division, Class, Student Name (one per row). Reload and everything syncs automatically.

function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // ── Save menus ──────────────────────────────────────────── if (data.type === 'menus') { var ms = ss.getSheetByName('Menus') || ss.insertSheet('Menus'); ms.clearContents(); ms.appendRow(['ID', 'Emoji', 'Name']); (data.menus || []).forEach(function(m) { ms.appendRow([m.id, m.emoji, m.name]); }); return ContentService.createTextOutput('ok'); } // ── Save students ───────────────────────────────────────── if (data.type === 'students') { var ss2 = SpreadsheetApp.getActiveSpreadsheet(); var st = ss2.getSheetByName('Students') || ss2.insertSheet('Students'); st.clearContents(); st.appendRow(['Division', 'Class', 'Student Name']); var students = data.students || {}; Object.keys(students).forEach(function(div) { var classes = students[div]; Object.keys(classes).forEach(function(cls) { (classes[cls] || []).forEach(function(name) { st.appendRow([div, cls, name]); }); }); }); return ContentService.createTextOutput('ok'); } // ── Log order ───────────────────────────────────────────── var date = data.date || new Date().toISOString().split('T')[0]; var time = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'HH:mm'); var sheet = ss.getSheetByName('Orders') || ss.insertSheet('Orders'); if (sheet.getLastRow() === 0) { sheet.appendRow(['Date','Time','Role','Division','Class','Name','Department','Meal','Quantity']); } if (data.role === 'teacher') { (data.orders || []).forEach(function(o) { sheet.appendRow([date, time, 'Teacher/Rep', data.division, data.cls || '', o.studentName, '', o.menuName, 1]); }); } else { sheet.appendRow([date, time, 'Staff', '', '', data.staffName, data.staffDept, data.menuName, data.quantity || 1]); } return ContentService.createTextOutput('ok'); } catch(err) { return ContentService.createTextOutput('error: ' + err.message); } } function doGet(e) { var action = e.parameter.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); var cb = e.parameter.callback || 'cb'; // ── Ping / warm-up ──────────────────────────────────────── if (action === 'ping') { return ContentService.createTextOutput(cb + '({"ok":true})').setMimeType(ContentService.MimeType.JAVASCRIPT); } // ── Load menus from the Menus sheet ────────────────────────── if (action === 'menus') { var sheet = ss.getSheetByName('Menus'); if (!sheet || sheet.getLastRow() < 2) { return ContentService.createTextOutput(cb + '({"menus":[]})').setMimeType(ContentService.MimeType.JAVASCRIPT); } var rows = sheet.getDataRange().getValues().slice(1); var menus = rows.filter(function(r){ return r[2]; }).map(function(r, i) { return { id: Number(r[0]) || (i+1), emoji: r[1] || '🍽', name: r[2].toString().trim() }; }); return ContentService.createTextOutput(cb + '(' + JSON.stringify({menus: menus}) + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } // ── Load students from the Students sheet ───────────────────── if (action === 'students') { var sheet = ss.getSheetByName('Students'); if (!sheet || sheet.getLastRow() < 2) { return ContentService.createTextOutput(cb + '({"students":{}})').setMimeType(ContentService.MimeType.JAVASCRIPT); } var rows = sheet.getDataRange().getValues().slice(1); var result = {}; rows.forEach(function(r) { var division = (r[0] || '').toString().trim(); var cls = (r[1] || '').toString().trim(); var name = (r[2] || '').toString().trim(); if (!division || !cls || !name) return; if (!result[division]) result[division] = {}; if (!result[division][cls]) result[division][cls] = []; result[division][cls].push(name); }); return ContentService.createTextOutput(cb + '(' + JSON.stringify({students: result}) + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } // ── Today's orders ──────────────────────────────────────────── if (action === 'orders') { var sheet = ss.getSheetByName('Orders'); if (!sheet || sheet.getLastRow() < 2) { return ContentService.createTextOutput(cb + '({"orders":[]})').setMimeType(ContentService.MimeType.JAVASCRIPT); } var today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); var rows = sheet.getDataRange().getValues(); var orders = rows.slice(1).filter(function(r){ return r[0] == today; }).map(function(r){ return { date:r[0], time:r[1], role:r[2], division:r[3], cls:r[4], name:r[5], dept:r[6], meal:r[7], qty:r[8] }; }); return ContentService.createTextOutput(cb + '(' + JSON.stringify({orders:orders}) + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } // ── Email summary ───────────────────────────────────────────── if (action === 'summary') { var recipient = 'admin@shieldschool.edu.gh'; // ← change this var sheet = ss.getSheetByName('Orders'); var today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); var rows = sheet ? sheet.getDataRange().getValues() : []; var todayRows = rows.slice(1).filter(function(r){ return r[0] == today; }); var body = 'Lunch orders for ' + today + ':\n\n'; todayRows.forEach(function(r){ body += r[2]+' | '+r[3]+' '+r[4]+' | '+r[5]+' | '+r[7]+(r[8]>1?' x'+r[8]:'')+'\n'; }); MailApp.sendEmail(recipient, 'Shield Lunch Orders – ' + today, body); return ContentService.createTextOutput(cb + '({"ok":true})').setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(cb + '({})').setMimeType(ContentService.MimeType.JAVASCRIPT); }
